# 歌声渲染流程图

```mermaid
graph TD
    RenderCall["RENDER action 调用"]
    RenderCall --> RendererMissing{{"SongTrackRenderer 是否未创建？"}}
    RendererMissing -- Yes --> CreateRenderer["创建 SongTrackRenderer 并注册事件监听器"]
    RendererMissing -- No --> LoopRunning{{"渲染循环是否正在执行？"}}
    CreateRenderer --> LoopRunning
    LoopRunning -- Yes --> ReRenderRequest["请求重新渲染（中断并重新开始）"]
    ReRenderRequest --> RenderEnd["RENDER action 结束"]
    LoopRunning -- No --> StartRenderRequest["请求开始渲染"]
    StartRenderRequest --> CanStart{{"有渲染开始（重新开始）请求，且无渲染停止请求？"}}
    CanStart -- No --> RenderEnd

    subgraph RenderingLoop["渲染循环"]
        CanStart -- Yes --> TakeSnapshot["创建项目快照"]
        TakeSnapshot --> GeneratePhrases["从快照中的音符生成短语"]
        GeneratePhrases --> PhrasesGeneratedEvent["发布 PhrasesGeneratedEvent"]
        PhrasesGeneratedEvent --> FilterRenderablePhrases["提取可渲染的短语"]
        FilterRenderablePhrases --> HasUnprocessedPhrases{{"是否存在未处理的短语？"}}

        subgraph LoadCache["应用缓存"]
            HasUnprocessedPhrases -- Yes --> GetUnprocessedPhrase["获取一个未处理的短语"]
            GetUnprocessedPhrase --> HasQueryCache{{"查询缓存是否存在？"}}
            HasQueryCache -- No --> NextPhrase["进入下一个短语"]
            HasQueryCache -- Yes --> LoadQueryCache["加载查询缓存并应用于短语"]

            LoadQueryCache --> HasPitchCache{{"歌声音高缓存是否存在？"}}
            HasPitchCache -- No --> NextPhrase
            HasPitchCache -- Yes --> LoadPitchCache["加载歌声音高缓存并应用于短语"]

            LoadPitchCache --> HasVolumeCache{{"歌声音量缓存是否存在？"}}
            HasVolumeCache -- No --> NextPhrase
            HasVolumeCache -- Yes --> LoadVolumeCache["加载歌声音量缓存并应用于短语"]

            LoadVolumeCache --> HasVoiceCache{{"歌声缓存是否存在？"}}
            HasVoiceCache -- No --> NextPhrase
            HasVoiceCache -- Yes --> LoadVoiceCache["加载歌声缓存并应用于短语"]

            LoadVoiceCache --> NextPhrase
            NextPhrase --> HasUnprocessedPhrases
        end

        HasUnprocessedPhrases -- No --> CacheLoadedEvent["发布 CacheLoadedEvent"]
        CacheLoadedEvent --> FilterPhrasesRequiringRender["提取需要渲染的短语"]
        FilterPhrasesRequiringRender --> CanRender{{"存在未处理的短语，且无中断请求？"}}

        subgraph RenderPhrases["渲染各个短语"]
            CanRender -- Yes --> SelectPhrase["选择要渲染的短语"]
            SelectPhrase --> PhraseRenderingStartedEvent["发布 PhraseRenderingStartedEvent"]

            subgraph RenderPhrase[" "]
                PhraseRenderingStartedEvent --> QueryNotGenerated{{"查询是否未生成？（无缓存？）"}}
                QueryNotGenerated -- Yes --> GenerateQuery["生成查询"]
                GenerateQuery --> CacheGeneratedQuery["将生成的查询保存至缓存"]
                QueryNotGenerated -- No --> PitchNotGenerated{{"歌声音高是否未生成？（无缓存？）"}}

                CacheGeneratedQuery --> QueryGenerationCompleteEvent["发布 QueryGenerationCompleteEvent"]
                QueryGenerationCompleteEvent --> PitchNotGenerated

                PitchNotGenerated -- Yes --> GeneratePitch["生成歌声音高"]
                GeneratePitch --> CacheGeneratedPitch["将生成的歌声音高保存至缓存"]
                PitchNotGenerated -- No --> VolumeNotGenerated{{"歌声音量是否未生成？（无缓存？）"}}

                CacheGeneratedPitch --> PitchGenerationCompleteEvent["发布 PitchGenerationCompleteEvent"]
                PitchGenerationCompleteEvent --> VolumeNotGenerated

                VolumeNotGenerated -- Yes --> GenerateVolume["生成歌声音量"]
                GenerateVolume --> CacheGeneratedVolume["将生成的歌声音量保存至缓存"]
                VolumeNotGenerated -- No --> VoiceNotGenerated{{"歌声是否未生成？（无缓存？）"}}

                CacheGeneratedVolume --> VolumeGenerationCompleteEvent["发布 VolumeGenerationCompleteEvent"]
                VolumeGenerationCompleteEvent --> VoiceNotGenerated

                VoiceNotGenerated -- Yes --> SynthesizeVoice["合成歌声"]
                SynthesizeVoice --> CacheSynthesizedVoice["将合成的歌声保存至缓存"]
                VoiceNotGenerated -- No --> PhraseRenderingCompleteEvent["发布 PhraseRenderingCompleteEvent"]

                CacheSynthesizedVoice --> VoiceSynthesisCompleteEvent["发布 VoiceSynthesisCompleteEvent"]
                VoiceSynthesisCompleteEvent --> PhraseRenderingCompleteEvent
            end

            PhraseRenderingErrorEvent["发布 PhraseRenderingErrorEvent"]
            PhraseRenderingCompleteEvent --> CanRender
            PhraseRenderingErrorEvent --> CanRender
            RenderPhrase -- 错误 --> PhraseRenderingErrorEvent
        end

        CanRender -- No --> CanStart
    end
```
